 * lấy danh sách cuộc trò chuyện
 - Mẫu data gửi lên:
 {
   "code": "room.getList",
   "data": {
     "size": 20,
     "page": 1
   }
 }



 * tạo cuộc trò chuyện
 - Mẫu data gửi lên:
 {
   "code": "room.createRoom",
   "data": {
     "type": "person", <--loại nhóm cá nhân(person) hoặc nhóm(group)-->
     "lstMemberId": [<--danh sách id thành viên nhóm trừ id của bản thân-->]
   }
 }

 - Mẫu data nhận được:
 {
   "statusCode": 200,
   "errorMessage": null,
   "code": "action",
   "data": {
     "id": <--id của room chat-->,
     "type": <--loại nhóm person/group-->,
     "name": <--tên nhóm-->,
     "userCreate": <--thông tin người tạo nhóm-->,
     "lstMember": <--danh sách thành viên, nếu isAdmin = true thì là có quyền admin-->,
     "createTime": <--thời gian tạo nhóm-->,
   }
 }

 * xóa cuộc trò chuyện
 - Mẫu data gửi lên:
 {
   "code": "room.deleteRoom",
   "data": {
     "id": <--id room-->
   }
 }

 * cập nhật thông tin cuộc trò chuyện
 - Mẫu data gửi lên:
 {
   "code": "room.updateRoom",
   "data": {
     "id": <--id room-->,
     "name": <--tên room mới-->,
   }
 }

 * gửi tin nhắn

   "code": "message.sendMessage",
   "data": {
     "id": <--id room-->,
     "content": <--nội dung tin nhắn-->,
     "lstFile": <--danh sách file ở dạng base64-->,
